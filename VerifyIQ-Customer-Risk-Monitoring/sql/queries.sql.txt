QUERY 1: Total customers

SELECT COUNT(*) AS total_clients
FROM clients;

QUERY 2: Customer type distribution

SELECT
    client_type,
    COUNT(*) AS total_clients
FROM clients
GROUP BY client_type;

QUERY 3: Risk distribution

SELECT
    sector_risk,
    COUNT(*) AS total_clients
FROM clients
GROUP BY sector_risk;

QUERY 4: PEP customers

SELECT COUNT(*) AS pep_clients
FROM clients
WHERE pep_flag = 1;

QUERY 5: Sanctioned customers

SELECT COUNT(*) AS sanctioned_clients
FROM clients
WHERE sanctions_flag = 1;

QUERY 6: FATF risky customers

SELECT COUNT(*) AS fatf_risk_clients
FROM clients
WHERE fatf_country_flag = 1;

QUERY 7: OFAC risky customers

SELECT COUNT(*) AS ofac_risk_clients
FROM clients
WHERE ofac_country_flag = 1;

QUERY 8: Total transactions

SELECT COUNT(*) AS total_transactions
FROM transactions;

QUERY 9: Total money moved

SELECT SUM(amount) AS total_amount
FROM transactions;

QUERY 10: Average transaction amount

SELECT AVG(amount) AS avg_transaction
FROM transactions;

QUERY 11: Transaction type distribution

SELECT
    transaction_type,
    COUNT(*) AS total_transactions
FROM transactions
GROUP BY transaction_type;

QUERY 12: Most active client countries

SELECT
    client_country,
    COUNT(*) AS total_transactions
FROM transactions
GROUP BY client_country
ORDER BY total_transactions DESC;

QUERY 13: OFAC suspicious transactions

SELECT COUNT(*) AS ofac_hits
FROM transactions
WHERE ofac_match_flag = 1;

QUERY 14: Structuring fraud cases

SELECT COUNT(*) AS structuring_cases
FROM transactions
WHERE structuring_pattern_flag = 1;

QUERY 15: Rapid suspicious movement

SELECT COUNT(*) AS rapid_cases
FROM transactions
WHERE rapid_movement_flag = 1;

QUERY 16: Trade mispricing fraud

SELECT COUNT(*) AS trade_cases
FROM transactions
WHERE trade_mispricing_flag = 1;

QUERY 17: High-risk customer money movement

SELECT
    c.sector_risk,
    SUM(t.amount) AS total_amount
FROM transactions t
JOIN clients c
ON t.client_id = c.client_id
GROUP BY c.sector_risk;

QUERY 18: PEP customer transaction volume

SELECT
    SUM(t.amount) AS pep_transaction_amount
FROM transactions t
JOIN clients c
ON t.client_id = c.client_id
WHERE c.pep_flag = 1;

QUERY 19: Sanctioned customer activity

SELECT
    COUNT(*) AS sanctioned_activity
FROM transactions t
JOIN clients c
ON t.client_id = c.client_id
WHERE c.sanctions_flag = 1;

QUERY 20: Top 10 customers by transaction amount

SELECT
    c.client_name,
    SUM(t.amount) AS total_amount
FROM transactions t
JOIN clients c
ON t.client_id = c.client_id
GROUP BY c.client_name
ORDER BY total_amount DESC
LIMIT 10;

QUERY 21: High-risk suspicious structuring

SELECT
    COUNT(*) AS suspicious_high_risk
FROM transactions t
JOIN clients c
ON t.client_id = c.client_id
WHERE c.sector_risk = 'High'
AND t.structuring_pattern_flag = 1;
